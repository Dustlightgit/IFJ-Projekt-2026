#include "ast.h"

int run_semantic = 1;

Program *build_ast(void) {
    Program *program = program_new();

    // def add(a: Int, b: Int): Int = return a + b
    FuncDef *add = funcdef_new("add", false, 1);
    funcdef_add_param(add, "a", type_new(TYPE_INT));
    funcdef_add_param(add, "b", type_new(TYPE_INT));
    funcdef_set_return_type(add, type_new(TYPE_INT));

    Block *add_body = block_new();
    Expr *sum = expr_new_binary(BINOP_ADD, expr_new_variable("a", 2),
                                expr_new_variable("b", 2), 2);
    block_add(add_body, stmt_new_return(sum, 2));
    funcdef_set_body(add, add_body);
    program_add_func(program, add);

    // @main def run(): Unit
    FuncDef *run = funcdef_new("run", true, 4);
    funcdef_set_return_type(run, type_new(TYPE_UNIT));

    Block *run_body = block_new();

    // val x: Int = add(1, 2)
    Expr *call_add = expr_new_call("add", 5);
    expr_call_add_arg(call_add, expr_new_int(1, 5));
    expr_call_add_arg(call_add, expr_new_int(2, 5));
    block_add(run_body,
              stmt_new_vardef(false, "x", type_new(TYPE_INT), call_add, 5));

    // print(x)
    Expr *call_print = expr_new_call("print", 6);
    expr_call_add_arg(call_print, expr_new_variable("x", 6));
    block_add(run_body, stmt_new_expr(call_print, 6));

    funcdef_set_body(run, run_body);
    program_add_func(program, run);

    return program;
}
